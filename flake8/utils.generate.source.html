253 def build_conf(spec):
254 name = spec['name']
255 data = test_data_names(spec)
256 lines = [f'{name}:', ' paths:']
257 for key, filename in data.items():
258 lines.append(f' {key}: file:test_data_dir/workflow/{filename}')
259 for output in spec['outputs']:
260 key = output[0]
261 # Some outputs are too big or depend on the run (e.g. they include paths), so they have no reference
262 if output[1] == 'file' and not spec.get('no_reference'):
263 lines.append(f' ref_{key}: file:test_reference_dir/workflow/{reference_name(spec, output)}')
264 lines.append(f' {key}: {output_name(spec, output)}')
265 lines.append(' properties:')
266 lines.append(' remove_tmp: true')
267 for key, value in spec.get('test_props', {}).items():
268 lines.append(f' {key}: {value}')
269 return '\n'.join(lines) + '\n'
